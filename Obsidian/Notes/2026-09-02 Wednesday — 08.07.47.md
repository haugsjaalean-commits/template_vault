		self._nom:str = ""
		self._marque:str = ""
		self.__no_serie:str = ""
	
	@property
	def no_serie(self) -> str:
		return self.__no_serie
		
	@no_serie.setter
	def no_serie(self, value:str) -> None:
		if not isinstance(value, str):
			raise TypeError("...")
